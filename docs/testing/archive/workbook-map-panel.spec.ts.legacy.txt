import { expect, test } from "@playwright/test";

test("지도 선택 패널과 접힌 도구는 이야기 왕복·뒤로가기에도 방문 상태를 유지한다", async ({
  page,
}) => {
  await page.emulateMedia({ reducedMotion: "reduce" });
  await page.goto("/workbook/classes/vibe-web");
  await page.getByRole("button", { name: "강의 시작하기 ↗" }).click();
  await page.getByRole("button", { name: /아직 모르겠어요 · 예시를 보며 시작할게요/ }).click();
  await page.getByRole("button", { name: "단원 지도 보기", exact: true }).click();
  const panel = page.getByRole("complementary", { name: "학습 패널" });
  const map = page.getByRole("region", { name: "카피바라의 AI 입문 지도" });
  await expect(map).toBeVisible();
  await expect(map.getByRole("button")).toHaveCount(6);
  await expect(map.getByRole("button", { name: /^AI 작업 이어가기/ })).toBeDisabled();
  await expect(page.getByText("학습 장소와 확인 기록", { exact: true })).toHaveCount(0);
  await expect(page.getByText("지도 사용 안내", { exact: true })).toHaveCount(0);
  await map
    .getByRole("button", { name: /AI 이해하기 (장소로 이동|학습 패널 열기)/, exact: true })
    .click();
  const selection = page.getByLabel("선택한 학습 장소");
  await expect(selection).toBeVisible();
  const topics = selection.locator("details").filter({ hasText: "이곳에서 다루는 주제" });
  await topics.locator("summary").press("Enter");
  await expect(topics).toContainText("1-1");
  await expect(topics).toContainText("생성형 AI · 초안과 사실 확인");
  await topics.locator("summary").press("Space");
  await selection.getByRole("button", { name: "선택 접기" }).press("Enter");
  await expect(selection).toBeHidden();
  await expect(
    map.getByRole("button", { name: "AI 이해하기 학습 패널 열기", exact: true }),
  ).toBeFocused();
  await map.getByRole("button", { name: "AI 이해하기 학습 패널 열기", exact: true }).click();
  await expect(selection).toBeVisible();
  await selection.getByRole("button", { name: "AI 이해하기 수업 시작" }).click();
  await page.getByRole("button", { name: "다음으로", exact: true }).click();
  await page.getByRole("button", { name: "단원 지도 보기", exact: true }).click();
  await expect(selection).toBeVisible();
  await selection.getByRole("button", { name: "AI 이해하기 수업 시작" }).click();
  await expect(
    page.getByText("이 클래스에서는 생성형 AI를 사용해요.", { exact: true }),
  ).toBeVisible();
  await page.getByRole("button", { name: "지금 할 일 ↑", exact: true }).click();
  await expect(page.locator("[data-current-action]")).toBeFocused();
  await page.getByRole("button", { name: "단원 지도 보기", exact: true }).click();
  await expect(
    map.getByRole("button", { name: "AI 종류와 도구 · 첫 수업 후 열려요" }),
  ).toBeDisabled();
  for (let repeat = 0; repeat < 2; repeat++) {
    await panel.getByText("학습 정보와 도구", { exact: true }).click();
    await panel
      .getByRole("button", { name: "카피바라 이야기 더 보기 · 선택", exact: true })
      .click();
    await page.getByRole("button", { name: "이야기 건너뛰고 학습 지도로 →", exact: true }).click();
    await expect(selection).toBeVisible();
    await expect(panel.getByText("0 / 5 학습 확인", { exact: true })).toHaveCount(0);
    await expect(
      map.getByRole("button", { name: "AI 이해하기 학습 패널 열기", exact: true }),
    ).toBeVisible();
  }
  await expect(page.getByRole("button", { name: "경로 움직임 멈추기", exact: true })).toHaveCount(
    0,
  );
  await page.getByRole("link", { name: "← 클래스 목록", exact: true }).click();
  await expect(page).toHaveURL(/\/workbook$/);
  await page.goBack();
  await expect(
    page.getByRole("button", { name: "AI 이해하기 수업 시작", exact: true }),
  ).toBeVisible();
  await expect(page.getByText("0 / 5 학습 확인", { exact: true })).toBeVisible();
});

test("첫 강의는 첫 행동·확인 조건·완료 뒤 다음 장소를 설명한다", async ({ page }) => {
  await page.emulateMedia({ reducedMotion: "reduce" });
  await page.goto("/workbook/classes/vibe-web");
  await page.getByRole("button", { name: "강의 시작하기 ↗" }).click();
  await page.getByRole("button", { name: /아직 모르겠어요 · 예시를 보며 시작할게요/ }).click();
  await page.getByRole("button", { name: "단원 지도 보기", exact: true }).click();
  await page
    .getByRole("button", { name: /AI 이해하기 (장소로 이동|학습 패널 열기)/, exact: true })
    .click();
  await page.getByRole("button", { name: "AI 이해하기 수업 시작", exact: true }).click();
  const action = page.locator("#workbook-lesson-next-action");
  await expect(action).toHaveText("다음으로에서 생성형 AI가 만드는 결과를 살펴봐요.");
  const next = page.getByRole("button", { name: "다음으로", exact: true });
  for (let part = 0; part < 3; part++) await next.click();
  await expect(action).toHaveText("예시 결과 보기를 눌러 소개문을 열어요.");
  await expect(next).toBeDisabled();
  await page.getByRole("button", { name: "예시 결과 보기", exact: true }).click();
  await expect(action).toHaveText("소개문에서 요청에 없던 사실을 하나 골라요.");
  await page.getByRole("button", { name: "차·호흡·돌 쌓는 웹 경험", exact: true }).click();
  await expect(next).toBeDisabled();
  await page.getByRole("button", { name: "이미 1만 명이 선택했다는 내용", exact: true }).click();
  await expect(action).toHaveText("다음으로에서 배운 내용을 정리해요.");
  await next.click();
  await expect(action).toContainText("AI 종류와 도구로 이어갈 수 있어요.");
  await page.getByRole("button", { name: "배운 내용 챙겨 마을로 →", exact: true }).click();
  await expect(page.getByText("1 / 5 학습 확인", { exact: true })).toBeVisible();
  await expect(
    page.getByRole("button", { name: "AI 종류와 도구 장소로 이동", exact: true }),
  ).toBeEnabled();
  await page.getByRole("button", { name: "클래스 소개 보기", exact: true }).click();
  await page.getByRole("button", { name: "학습으로 돌아가기 ↗", exact: true }).click();
  await page.getByRole("button", { name: "단원 지도 보기", exact: true }).click();
  await expect(page.getByText("1 / 5 학습 확인", { exact: true })).toBeVisible();
  await expect(
    page.getByRole("button", { name: "AI 종류와 도구 학습 패널 열기", exact: true }),
  ).toBeVisible();
});

test("홈에서 클래스의 구성 예시를 보고 목적을 선택하고 변경·소개 왕복·뒤로가기에도 유지한다", async ({
  page,
}) => {
  await page.goto("/workbook");
  await expect(page.locator("#results")).toHaveCount(0);
  await page.getByRole("link", { name: "웹페이지 만들기 클래스 보기 →", exact: true }).click();
  for (const id of ["service", "portfolio", "personal"]) {
    const card = page.locator(`#example-${id}`);
    const image = card.getByRole("img");
    await image.scrollIntoViewIfNeeded();
    await expect(image).toHaveAttribute("src", new RegExp(`${id}-website-v2`));
    await expect(image).toHaveAttribute("alt", /사이트.*디자인 예시/);
    await expect(image).toHaveAttribute("loading", "lazy");
    await expect
      .poll(() => image.evaluate((el: HTMLImageElement) => el.complete && el.naturalWidth > 0))
      .toBe(true);
    await expect(card).toContainText("구성 예시");
  }
  await expect(page).toHaveURL(/classes\/vibe-web$/);
  await expect(page.locator("#example-portfolio")).toBeVisible();
  await expect(page.getByText("선택한 클래스 · 바이브코딩 웹개발", { exact: true })).toBeVisible();
  await expect(
    page.getByRole("heading", { name: /내 콘텐츠를 담은.*작은 웹페이지 하나/ }),
  ).toBeVisible();
  const introOrder = await page
    .locator("#class-curriculum-title, #class-results-title")
    .allTextContents();
  expect(introOrder).toEqual(["다루는 내용과 커리큘럼", "내 목적에 맞게, 첫 화면을 만들어요"]);
  const curriculum = page.getByRole("region", { name: "다루는 내용과 커리큘럼" });
  await curriculum.locator("summary").filter({ hasText: "CSS와 반응형 웹" }).press("Enter");
  await expect(
    curriculum.getByText("반응형 웹 · 모바일과 데스크톱 비교", { exact: true }),
  ).toBeVisible();
  await curriculum.locator("summary").filter({ hasText: "CSS와 반응형 웹" }).press("Space");
  await expect(
    curriculum.getByText("반응형 웹 · 모바일과 데스크톱 비교", { exact: true }),
  ).toBeHidden();
  await expect(curriculum).toContainText("실전 심화");
  await expect(curriculum).toContainText("별도 월드 ‘AI와 일하는 방식’은 준비 중");
  await expect(page.locator("#example-portfolio").getByRole("img")).toHaveAttribute(
    "src",
    /portfolio-website-v2/,
  );
  await page.getByRole("button", { name: "강의 시작하기 ↗", exact: true }).click();
  await expect(
    page.getByRole("heading", { name: "어떤 페이지를 만들어보고 싶나요?" }),
  ).toBeVisible();
  await page.getByRole("button", { name: /내 작업과 경험을 보여주고 싶어요/ }).click();
  await page.getByRole("button", { name: "단원 지도 보기", exact: true }).click();
  const panel = page.getByRole("complementary", { name: "학습 패널" });
  await expect(panel).toContainText("내 작업을 보여주는 포트폴리오 만들기");
  await panel.getByRole("button", { name: "목적 바꾸기", exact: true }).click();
  await page.getByRole("button", { name: /작은 서비스를 소개하고 싶어요/ }).click();
  await expect(panel).toContainText("내 아이디어를 소개하는 페이지 만들기");
  await page.getByRole("button", { name: "클래스 소개 보기", exact: true }).click();
  await page.getByRole("button", { name: "학습으로 돌아가기 ↗", exact: true }).click();
  await page.getByRole("button", { name: "단원 지도 보기", exact: true }).click();
  await expect(panel).toContainText("내 아이디어를 소개하는 페이지 만들기");
  await page.getByRole("link", { name: "← 클래스 목록", exact: true }).click();
  await expect(page).toHaveURL(/\/workbook$/);
  await page.goBack();
  await expect(panel).toContainText("내 아이디어를 소개하는 페이지 만들기");
});

for (const purpose of [
  {
    choice: "작은 서비스를 소개하고 싶어요",
    caution: "아직 제공하지 않는 기능이나 확인하지 않은 고객 반응",
    goal: "내 아이디어를 소개하는 페이지 만들기",
  },
  {
    choice: "내 작업과 경험을 보여주고 싶어요",
    caution: "직접 하지 않은 역할이나 확인하지 않은 성과 수치",
    goal: "내 작업을 보여주는 포트폴리오 만들기",
  },
  {
    choice: "좋아하는 것으로 사이트를 만들고 싶어요",
    caution: "방문하지 않은 장소나 해보지 않은 경험",
    goal: "좋아하는 것을 담은 개인 사이트 만들기",
  },
]) {
  test(`${purpose.choice}: 수업에서 목적별 확인 기준을 보고 학습 정리를 챙긴다`, async ({
    page,
  }) => {
    await page.emulateMedia({ reducedMotion: "reduce" });
    await page.goto("/workbook/classes/vibe-web");
    await page.getByRole("button", { name: "강의 시작하기 ↗", exact: true }).click();
    await page.getByRole("button", { name: new RegExp(purpose.choice) }).click();
    await expect(page.getByRole("heading", { name: "AI 이해하기", exact: true })).toBeFocused();
    await page.getByText("내 목적과 연결하기", { exact: true }).click();
    await expect(page.getByRole("heading", { name: purpose.goal, exact: true })).toBeVisible();
    await expect(page.getByText(purpose.caution, { exact: false }).first()).toBeVisible();
    for (let i = 0; i < 3; i++)
      await page.getByRole("button", { name: "다음으로", exact: true }).click();
    await page.getByRole("button", { name: "예시 결과 보기", exact: true }).click();
    await page.getByRole("button", { name: "이미 1만 명이 선택했다는 내용", exact: true }).click();
    await page.getByRole("button", { name: "다음으로", exact: true }).click();
    const recap = page.getByRole("region", { name: "배운 내용과 내 작업의 변화", exact: true });
    await expect(recap).toContainText(purpose.caution);
    await expect(recap).toContainText("아직 내 파일을 제작하거나 수정한 단계는 아니에요.");
    await page.getByRole("button", { name: "배운 내용 챙겨 마을로 →", exact: true }).click();
    await expect(page.getByText("1 / 5 학습 확인", { exact: true })).toBeVisible();
  });
}
