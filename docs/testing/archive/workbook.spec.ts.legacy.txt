async function visitPractical(page: Page, stage: PracticalStage) {
  const name = practicalMapStops[stage].name;
  const mapNames = {
    agent: "카피바라의 AI 입문 지도",
    project: "기획과 첫 제작 지도",
    experience: "화면과 사용 경험 지도",
    data: "동작과 데이터 지도",
    publish: "공개와 운영 지도",
  };
  const map = page.getByRole("region", { name: mapNames[stage], exact: true });
  await expect(map.getByRole("button")).toHaveCount(stage === "agent" ? 6 : 4);
  await expect(map.locator('polyline[data-branch="true"]')).toHaveCount(1);
  const point = page.getByRole("button", { name: `${name} 선택`, exact: true });
  await expect(point).toBeEnabled();
  await point.press("Enter");
  const panel = page.getByRole("region", { name: "선택한 심화 장소", exact: true });
  await expect(panel.getByRole("heading")).toHaveText(name);
  await panel.getByRole("button", { name: `${name} 수업 시작`, exact: true }).press("Enter");
  const lesson = page.getByRole("region", { name: "실전 심화 수업", exact: true });
  await expect(lesson.getByRole("heading", { level: 1 })).toBeFocused();
  const situation = practicalLabs[stage].situations[0];
  await lesson.getByRole("button", { name: `1. ${situation.name}`, exact: true }).click();
  await lesson
    .getByRole("button", { name: situation.options[situation.best], exact: true })
    .click();
  await expect(lesson).toContainText(situation.check);
  if (stage === "data") {
    await page.getByRole("link", { name: /^workbook$/i, exact: true }).click();
    await expect(page).toHaveURL(/\/workbook$/);
    await page.goBack();
    await expect(lesson.locator('[data-result="correct"]')).toHaveAttribute("aria-pressed", "true");
  }
  await lesson.getByRole("button", { name: "같은 지도로 돌아가기 →", exact: true }).click();
  await expect(point).toBeFocused();
  await expect(panel).toContainText("1 / 3 상황 살펴봄");
  await panel.getByRole("button", { name: `${name} 수업 시작`, exact: true }).click();
  await expect(lesson.locator('[data-result="correct"]')).toHaveAttribute("aria-pressed", "true");
  await lesson.getByRole("button", { name: "단원 지도 보기", exact: true }).click();
  await panel.getByRole("button", { name: "선택 접기", exact: true }).press("Enter");
  await expect(point).toBeFocused();
  await openRecords(page); // 목록 왕복 후 접힌 기록 패널을 다시 열어 다음 검사를 이어간다.
}

async function openRecords(page: Page) {
  for (const name of ["학습·제작 기록 확인"]) {
    const summary = page.getByText(name, { exact: true });
    if (await summary.isVisible()) {
      const details = summary.locator("..");
      if ((await details.getAttribute("open")) === null) await summary.click();
    }
  }
}
async function openStory(page: Page) {
  const summary = page.getByText("학습 정보와 도구", { exact: true });
  if ((await summary.locator("..").getAttribute("open")) === null) await summary.click();
  await page.getByRole("button", { name: "카피바라 이야기 더 보기 · 선택", exact: true }).click();
}
async function openMapLesson(page: Page, name: string, key?: string) {
  const point = page.getByRole("button", { name: `${name} 선택`, exact: true });
  if (key) await point.press(key);
  else await point.click();
  await expect(
    page
      .getByLabel("선택한 학습 장소")
      .locator("summary")
      .filter({ hasText: "이곳에서 다루는 주제" }),
  ).toBeVisible();
  const panel = page.getByLabel("선택한 학습 장소");
  // 지점 선택이 예약한 시작 버튼 포커스가 도착한 뒤 다음 키보드 조작을 한다.
  await expect(panel.getByRole("button", { name: `${name} 수업 시작`, exact: true })).toBeFocused();
  await panel.getByRole("button", { name: "선택 접기", exact: true }).press("Enter");
  await expect(point).toBeFocused();
  await point.press("Enter");
  await page.getByRole("button", { name: `${name} 수업 시작`, exact: true }).click();
}

async function chooseMap(page: Page, name: string, keyboard = false) {
  const journey = page.getByTestId("workbook-journey");
  if ((await journey.getAttribute("open")) === null)
    await page.getByRole("button", { name: "전체 과정 보기", exact: true }).click();
  const button = journey.getByRole("button", { name, exact: true, includeHidden: true });
  const card = button.locator("..").locator("..");
  if ((await card.getAttribute("open")) === null) await card.locator("summary").click();
  if (keyboard) await button.press("Enter");
  else await button.click();
}
